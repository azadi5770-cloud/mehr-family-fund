function $(id){return document.getElementById(id)}
function qa(s){return Array.from(document.querySelectorAll(s))}
function esc(v){return String(v??"").replace(/[&<>"']/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[m]))}
function nl2br(v){return esc(v).replace(/\n/g,"<br>")}
function normalizeDigits(str){return String(str??"").replace(/[۰-۹]/g,d=>"۰۱۲۳۴۵۶۷۸۹".indexOf(d)).replace(/[٠-٩]/g,d=>"٠١٢٣٤٥٦٧٨٩".indexOf(d))}
function toNumber(v){return Number(normalizeDigits(v).replace(/,/g,"").replace(/[^\d.-]/g,""))||0}
function formatInputNumber(v){let n=toNumber(v);return n?n.toLocaleString("en-US"):""}
function money(n){return (Number(n)||0).toLocaleString("en-US")+" "+app.settings.currency}
function todayJ(){
    const parts=new Intl.DateTimeFormat("fa-IR-u-ca-persian",{year:"numeric",month:"2-digit",day:"2-digit"}).formatToParts(new Date());
    const y=normalizeDigits(parts.find(p=>p.type==="year").value);
    const m=normalizeDigits(parts.find(p=>p.type==="month").value).padStart(2,"0");
    const d=normalizeDigits(parts.find(p=>p.type==="day").value).padStart(2,"0");
    return `${y}/${m}/${d}`;
}
function normalizeDateJ(v){
    let s=normalizeDigits(v).trim().replace(/-/g,"/").replace(/\./g,"/");
    const p=s.split("/");
    if(p.length>=3)return `${p[0].padStart(4,"0")}/${p[1].padStart(2,"0")}/${p[2].padStart(2,"0")}`;
    return s;
}
function dateKey(v){return normalizeDateJ(v).replace(/\D/g,"")||"00000000"}
function monthNumberJ(v){const p=normalizeDateJ(v).split("/").map(Number);return p.length>=2?p[0]*12+p[1]:0}
function signedTx(t){return (t.direction||(TX_TYPES[t.type]?.direction||"in"))==="out"?-(Number(t.amount)||0):(Number(t.amount)||0)}
function approvedTx(){return app.transactions.filter(t=>t.status==="approved")}
function fundBalance(){return approvedTx().reduce((s,t)=>s+signedTx(t),0)}
function totalIncome(){return approvedTx().filter(t=>signedTx(t)>0).reduce((s,t)=>s+signedTx(t),0)}
function totalOutflow(){return Math.abs(approvedTx().filter(t=>signedTx(t)<0).reduce((s,t)=>s+signedTx(t),0))}
function memberName(id,fallback="بدون عضو"){return app.members.find(m=>m.id===id)?.full_name || fallback}
function typeLabel(t){return TX_TYPES[t]?.label||t}
function statusBadge(st){const s=STATUS[st]||STATUS.pending;return `<span class="badge ${s.class}">${s.label}</span>`}
function loanPaidAmount(id,txs=app.transactions){return txs.filter(t=>t.loan_id===id&&t.type==="installment"&&t.status==="approved").reduce((s,t)=>s+Number(t.amount||0),0)}
function loanRemaining(l,txs=app.transactions){return Math.max(Number(l.principal||0)-loanPaidAmount(l.id,txs),0)}
function loanDueInfo(l,txs=app.transactions){
    const paid=loanPaidAmount(l.id,txs);
    const inst=Number(l.installment||0)||Math.ceil(Number(l.principal||0)/(Number(l.months)||1));
    const dueCount=Math.min(Number(l.months)||0,Math.max(0,monthNumberJ(todayJ())-monthNumberJ(l.date_j)));
    const dueAmount=Math.max(0,Math.min(inst*dueCount,Number(l.principal)||0)-paid);
    return {paid,remaining:Math.max(Number(l.principal||0)-paid,0),dueAmount};
}
function showToast(text,type="info"){
    const t=$("toast");
    t.textContent=text;
    t.style.background=type==="success"?"#166534":type==="danger"?"#991b1b":type==="warning"?"#92400e":"#0f172a";
    t.classList.add("show");
    setTimeout(()=>t.classList.remove("show"),4200);
}

async function loadSettings(){
    const {data,error}=await db.from("fund_settings").select("*").eq("id",1).single();
    if(error){showToast("خطا در خواندن تنظیمات: "+error.message,"danger");return}
    const extra=data.settings||{};
    app.settings={
        ...DEFAULT_SETTINGS,
        ...extra,
        fundName:data.fund_name||DEFAULT_SETTINGS.fundName,
        shortName:data.short_name||DEFAULT_SETTINGS.shortName,
        establishedYear:data.established_year||DEFAULT_SETTINGS.establishedYear,
        monthlyShare:Number(data.monthly_share||DEFAULT_SETTINGS.monthlyShare),
        currency:data.currency||DEFAULT_SETTINGS.currency,
        gatewayTitle:data.gateway_title||DEFAULT_SETTINGS.gatewayTitle,
        gatewayLink:data.gateway_link||"",
        gatewayEnabled:!!data.gateway_enabled
    };
    applySettings();
}

async function loadAdminData(){
    const [m,t,l,r]=await Promise.all([
        db.from("members").select("*").order("created_at",{ascending:false}),
        db.from("transactions").select("*").order("date_j",{ascending:false}),
        db.from("loans").select("*").order("date_j",{ascending:false}),
        db.from("loan_requests").select("*").order("date_j",{ascending:false})
    ]);
    if(m.error||t.error||l.error||r.error){

    app.transactions=t.data||[];
    app.loans=l.data||[];
    app.loanRequests=r.data||[];
    renderHomeKpis();
    return true;
}

function memberSummaryFromData(member,txs,loans){
    const share=txs.filter(t=>t.type==="share"&&t.status==="approved").reduce((s,t)=>s+Number(t.amount||0),0);
    const paidInstallments=txs.filter(t=>t.type==="installment"&&t.status==="approved").reduce((s,t)=>s+Number(t.amount||0),0);
    const remaining=loans.reduce((s,l)=>s+loanRemaining(l,txs),0);
    const overdue=loans.reduce((s,l)=>s+loanDueInfo(l,txs).dueAmount,0);
    return {share,paidInstallments,remaining,overdue,net:share-remaining};
}

async function saveLoan(e,id){
    e.preventDefault();
    const f=new FormData(e.target);
    const principal=toNumber(f.get("principal"));
    const months=Number(f.get("months"))||1;
    const row={
        member_id:f.get("member_id"),
        date_j:normalizeDateJ(f.get("date_j")),
        principal,
        months,
        fee_percent:toNumber(f.get("fee_percent")),
        installment:toNumber(f.get("installment"))||Math.ceil(principal/months),
        status:f.get("status"),
        description:f.get("description")||""
    };
    let loanId=id;
    if(id){
        const {error}=await db.from("loans").update(row).eq("id",id);
        if(error){showToast(error.message,"danger");return}
    }else{
        const {data,error}=await db.from("loans").insert(row).select().single();
        if(error){showToast(error.message,"danger");return}
        loanId=data.id;
        if(f.get("createTx")==="yes"){
            const mem=app.members.find(m=>m.id===row.member_id);
            await db.from("transactions").insert({
                date_j:row.date_j,member_id:row.member_id,member_name_snapshot:mem?.full_name||"",phone_snapshot:mem?.phone||"",
                type:"loan_disbursement",direction:"out",amount:principal,loan_id:loanId,status:"approved",description:"پرداخت وام"
            });
            const fee=Math.round(principal*row.fee_percent/100);
            if(fee>0)await db.from("transactions").insert({
                date_j:row.date_j,member_id:row.member_id,member_name_snapshot:mem?.full_name||"",phone_snapshot:mem?.phone||"",
                type:"fee",direction:"in",amount:fee,loan_id:loanId,status:"approved",description:"کارمزد وام"
            });
        }
    }
    await loadAdminData();closeModal();renderLoans();showToast("وام ذخیره شد.","success");
}

function accountingTable(){

    const rows=app.transactions.slice().sort((a,b)=>dateKey(a.date_j)-dateKey(b.date_j)).map(t=>{

        if(t.status==="approved")bal+=s;

async function saveSettings(e){
    e.preventDefault();
    const f=new FormData(e.target);
    const s={
        ...app.settings,
        fundName:f.get("fundName"),
        shortName:f.get("shortName"),
        establishedYear:f.get("establishedYear"),
        monthlyShare:toNumber(f.get("monthlyShare")),
        currency:f.get("currency"),
        bankCard:f.get("bankCard"),
        sheba:f.get("sheba"),
        accountOwner:f.get("accountOwner"),
        gatewayLink:f.get("gatewayLink"),
        gatewayTitle:f.get("gatewayTitle"),
        gatewayEnabled:f.get("gatewayEnabled")==="true",
        primary:f.get("primary"),
        secondary:f.get("secondary"),
        accent:f.get("accent"),
        homeTitle:f.get("homeTitle"),
        homeText:f.get("homeText"),
        announcement:f.get("announcement"),
        rules:f.get("rules")
    };
    const {error}=await db.from("fund_settings").update({
        fund_name:s.fundName,
        short_name:s.shortName,
        established_year:s.establishedYear,
        monthly_share:s.monthlyShare,
        currency:s.currency,
        gateway_title:s.gatewayTitle,
        gateway_link:s.gatewayLink,
        gateway_enabled:s.gatewayEnabled,
        settings:s,
        updated_at:new Date().toISOString()
    }).eq("id",1);
    if(error){showToast(error.message,"danger");return}
    app.settings=s;
    applySettings();
    showToast("تنظیمات ذخیره شد.","success");
}
async function uploadLogo(e){
    const file=e.target.files[0];
    if(!file)return;
    const reader=new FileReader();
    reader.onload=async()=>{
        app.settings.logoData=reader.result;
        await db.from("fund_settings").update({settings:app.settings}).eq("id",1);
        applySettings();
        showToast("لوگو ذخیره شد.","success");
    };
    reader.readAsDataURL(file);
}

function downloadCSV(name,csv){
    const blob=new Blob(["\ufeff"+csv],{type:"text/csv;charset=utf-8"});
    const a=document.createElement("a");
    a.href=URL.createObjectURL(blob);
    a.download=name;
    a.click();
    URL.revokeObjectURL(a.href);
}

function membersCSV(){return "code,full_name,phone,join_date_j,share_count,status\n"+app.members.map(m=>[m.code,m.full_name,m.phone,m.join_date_j,m.share_count,m.status].map(csv).join(",")).join("\n")}

function loansCSV(){return "date_j,member,principal,months,installment,paid,remaining,status\n"+app.loans.map(l=>{const d=loanDueInfo(l);return [l.date_j,memberName(l.member_id),l.principal,l.months,l.installment,d.paid,d.remaining,l.status].map(csv).join(",")}).join("\n")}
function accountingCSV(){
    let bal=0;
    return "date_j,type,member,income,outcome,balance,status\n"+app.transactions.slice().sort((a,b)=>dateKey(a.date_j)-dateKey(b.date_j)).map(t=>{
        const s=signedTx(t); if(t.status==="approved")bal+=s;
        return [t.date_j,typeLabel(t.type),memberName(t.member_id,t.member_name_snapshot),s>0?s:"",s<0?Math.abs(s):"",t.status==="approved"?bal:"",t.status].map(csv).join(",");
    }).join("\n");
}

    }else{
        window.print();
    }
}
